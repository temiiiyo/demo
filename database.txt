create database shveyka;
use shveyka;

create table users (id int primary key auto_increment, FIO varchar(255), login varchar(255), password varchar(255), role varchar(255));
create table proizvoditel ( id int primary key auto_increment, name varchar(255));
create table postavchik (id int primary key auto_increment, name varchar(255));
create table category (id int primary key auto_increment, name varchar(255));
create table product (id int primary key auto_increment, article varchar(255) unique, name varchar(255), izmerenie varchar(255), price decimal(8,2), max_discount int,
proizvoditel_id int,
postavchik_id int,
category_id int,
current_discount int,
quantity int,
description text,
photo_path varchar(255),
foreign key(proizvoditel_id) references proizvoditel(id),
foreign key(postavchik_id) references postavchik(id),
foreign key(category_id) references category(id)
);

create table pvz (id int primary key auto_increment, postcode varchar(255), city varchar(255), street varchar(255), house varchar(255));
create table orders (id int primary key auto_increment, order_date date, delivery_date date,
pvz_id int,
user_id int,
pvz_code varchar(255),
status varchar(255),
foreign key(pvz_id) references pvz(id),
foreign key(user_id) references users(id));

create table orders_items (id int primary key auto_increment, orders_id int, product_id int, quantity int, price decimal(8,2),
foreign key(orders_id) references orders(id),
foreign key(product_id) references product(id));

select *from product;

select c.name, p.name, pr.name, p.price
        from product p
        join category c on c.id = p.category_id
        join proizvoditel pr on pr.id = p.proizvoditel_id
        where 1 = 1;

select *from orders;
select *from orders_items;


select o.id, o.order_date, o.status,
sum(p.price * oi.quantity) as total
from product p
join orders_items oi on oi.product_id = p.id
join orders o on o.id = oi.orders_id
group by o.id
order by o.order_date;



select p.id, c.name as cat, p.name as p_name, pr.name as pr_name, p.price, p.photo_path
